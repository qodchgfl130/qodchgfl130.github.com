    def PushAndSerch(self):
        self.tableWidget.setRowCount(0)
        try:
            with self.conn.cursor() as cursor:

                sql = """
                SELECT n.newsday, n.comname, n.newsname, n.polarity, m.newsnumber , n.Category 
                FROM news n 
                JOIN mainnews m ON n.newsnumber = m.newsnumber 
                WHERE n.Category = %s AND n.ownday = %s;
                            """
                cursor.execute(sql, (self.selected_category, self.dateStr))
                self.result = cursor.fetchall()
                QPushButton('확인')


                if len(self.result) == 0:
                    QMessageBox.warning(self, '오류발생', '저장 되어있는 값이 없습니다')
                for no in range(len(self.result)):

                    day = self.result[no][0]
                    company = self.result[no][1]
                    name = self.result[no][2]
                    category = str(self.result[no][3])
                    number = str(self.result[no][4])

                    btn = QPushButton('본문가기')

                    self.tableWidget.setItem(no, 1, QTableWidgetItem(company))
                    self.tableWidget.setItem(no, 2, QTableWidgetItem(name))
                    self.tableWidget.setItem(no, 3, QTableWidgetItem(category))
                    self.tableWidget.setCellWidget(no, 4, btn)

        except Exception as e:

            QMessageBox.warning(self, '오류발생', f'DB 조회 중 오류 발생: {e}')

    def SerchCategory(self):

    def OnDateChanged(self):
        self.dateStr = self.dateEdit.date().toString('yyyyMMdd')

    def ShowTheMainNews(self, row):
        self.second = row
        try:
            with self.conn.cursor() as cursor:

                sql = 'SELECT main FROM mainnews WHERE newsnumber = %s;'
                cursor.execute(sql, (row,))
                result = cursor.fetchone()

                raw_text = result[0]
                self.textBrowser.setText(raw_text)

        except Exception as e:
            QMessageBox.warning(self, '오류발생', f'DB 조회 중 오류 발생: {e}')

    def SecondWindow(self):
